# Load JSON data from your existing file $json = Get-Content -Path "path_to_your_existing_json_file.json" | ConvertFrom-Json # Function to calculate clan stats function Get-ClanStats($clan) { $totalElo = 0 $totalWins = 0 $totalGames = 0 $playerCount = $clan.players.Count foreach ($player in $clan.players) { $totalElo += $player.elo $totalWins += $player.wins $totalGames += $player.games } $winPercentage = if ($totalGames -gt 0) { [math]::round(($totalWins / $totalGames) * 100, 2) } else { 0 } $averageElo = if ($playerCount -gt 0) { [math]::round($totalElo / $playerCount, 2) } else { 0 } return @{ "Name" = $clan.name "Logo" = $clan.logo "Win %" = $winPercentage "Average Elo" = $averageElo "Total Elo" = $totalElo "Games Played" = $totalGames "Ranking" = $null # To be calculated later "Players" = $clan.players } } # Get clan stats $clanStats = $json.clans | ForEach-Object { Get-ClanStats $_ } # Sort clans by total Elo (ranking) $sortedClans = $clanStats | Sort-Object -Property "Total Elo" -Descending # Assign rankings $ranking = 1 foreach ($clan in $sortedClans) { $clan.Ranking = $ranking $ranking++ } # Display the results $sortedClans | ForEach-Object { Write-Host "`nClan: $($_.Name)" Write-Host "Logo: $($_.Logo)" Write-Host "Win %: $($_.['Win %'])" Write-Host "Average Elo: $($_.['Average Elo'])" Write-Host "Total Elo: $($_.['Total Elo'])" Write-Host "Games Played: $($_.['Games Played'])" Write-Host "Ranking: $($_.Ranking)" Write-Host "Players in this Clan:" # Display players $_.Players | ForEach-Object { Write-Host " - $($_.name): Elo = $($_.elo), Wins = $($_.wins), Games Played = $($_.games)" } Write-Host "----------------------------------" }
